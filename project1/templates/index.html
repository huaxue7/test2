// ============================================================ // 排行榜（含 localStorage 持久化） // ============================================================ const LS_KEY = 'tetris_scores'; function getLocalScores() { try { return JSON.parse(localStorage.getItem(LS_KEY)) || []; } catch { return []; } } function saveLocalScores(scores) { try { localStorage.setItem(LS_KEY, JSON.stringify(scores)); } catch { /* ignore */ } } function mergeScores(remote, local) { const map = new Map(); for (const s of [...remote, ...local]) { const key = s.name + s.score + s.date; if (!map.has(key)) map.set(key, s); } return Array.from(map.values()).sort((a, b) => b.score - a.score); }
